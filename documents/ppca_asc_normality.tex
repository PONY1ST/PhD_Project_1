\chapter{Low Dimensional Asymptotic Normality}

\section{Basic Setup}\label{sec:basic_setup}

Given a "high dimensional" data set $\{\bm{x}_i\}_{i=1}^n$
where $\bm{x}_i \in \mathbb{R}^p$ for all $i = 1, \ldots, n$,
the statistician fixes a positive integer $q < p$ 
and wishes to find the $q$ principal components.
Assuming that the data is centred (that is, with mean $\bm{0}$),
the probabilistic principal component analysis model
is a statistical model that
treats $\bm{x}_i$'s as realizations 
of independent random variables $\bm{X}_i = \bm{W}\bm{Z}_i + \bm{\epsilon_i}$,
where $\bm{W}$ is an unknown $p \times q$ loading matrix of full rank, 
$\bm{Z}_i \sim N_q\bracket{\bm{0}, \bm{I}_q}$ is the latent normal variable,
and $\bm{\epsilon_i} \sim N_p\bracket{\bm{0}, \sigma^2\bm{I}_p}$
is the Gaussian noise with unknown standard deviation $\sigma > 0$,
and $(\bm{Z}_1, \ldots, \bm{Z}_n, \bm{\epsilon}_1, \ldots, \bm{\epsilon}_n)$
are all independent (for now). 
Put $\theta = (\bm{W}_{\theta}, \sigma_{\theta}^2)$ for the unknown 
parameter to estimate
and $\Theta$ for all possible choices of $\theta$.
$\Theta$ is thus the collection of all tuples $(\bm{W}, \sigma^2)$
such that $\bm{W}$ is a real $p \times q$ matrix with $\text{rank}(\bm{W}) = q$
and $\sigma^2 \in (0, \infty)$.
Given that the collection of all real $p \times q$ matrix with rank $q$
is an open subset of the Euclidean space $\mathbb{R}^{pq}$,
we consider $\Theta$ as an open subset of the Euclidean space $\mathbb{R}^{pq+1}$
and our statistical model is 

\begin{equation}\label{eq:original_model}
  \mathcal{P} = \curly{N_p\bracket{\bm{0}, \bm{W}_{\theta}\bm{W}_{\theta}^T + \sigma_{\theta}^2\bm{I}_p}}_{\theta \in \Theta}.
\end{equation}

To resolve the non-identifiability of $\mathcal{P}$,
we introduce the quotient manifold structure.
For $(\bm{A},s) \in \Theta$ and $(\bm{B}, t) \in \Theta$,
we define $(\bm{A},s) \sim (\bm{B}, t)$
to mean that they give rise to the same covariance matrix,namely when
$\bm{A}\bm{A}^T + s\bm{I}_p = \bm{B}\bm{B}^T + t\bm{I}_p$.
The equivalence classes are denoted by $[(\bm{A},s)]$
or more briefly $[\bm{A},s]$. 
Unless otherwise mentioned, $(\bm{A},s) \in \Theta$ is the default representative 
we pick when we write an equivalence class as $[\bm{A},s]$.
Writing $O(q)$ for the collection of all $q \times q$ real orthogonal matrices,
it was shown in Datta's thesis that 
$[\bm{A},s] = \{(\bm{A}\bm{Q},s) \mid \bm{Q} \in O(q)\}$.
Therefore, $M := \Theta / \sim$ can be identified as the orbits 
of the right action of $O(q)$ on $\Theta$
and, given that this Lie group action is smooth, proper, and free,
the quotient manifold theorem (see \cite{Lee_2013}, Theorem 21.10) shows that there is a unique smooth 
structure making $M$ a manifold of dimension $pq+1-\frac{q(q-1)}{2}$ and 
the map $\pi: \Theta \to M$ 
defined by

\begin{equation}\label{eq:pi_def}
  \pi((\bm{A},s)) = [\bm{A},s]
\end{equation}
is a smooth submersion.
Moreover, this action is verticle, transitive on fibers 
and is isomertic with respect to the Euclidean metric.
Thus, there exists a unique metric $g$ on $M$ making
$(M,g)$ a Riemannian manifold and $\pi$ a Riemannian submersion
(see \cite{Lee_2018}, Theorem 2.28), meaning that

\begin{equation}\label{eq:g_def}
  g_{(\bm{A},s)}^{\Theta}\bracket{(\bm{V},c), (\bm{U},b)}
  = g_{[\bm{A},s]} \bracket{d\pi_{(A,s)}((\bm{V},c)), d\pi_{(A,s)}((\bm{U},b))},
\end{equation}
where $g_{(\bm{A},s)}^{\Theta}$ is the standard Euclidean metric given by 
\begin{equation}\label{eq:euclidean_metric}
  g_{(\bm{A},s)}^{\Theta}\bracket{(\bm{V},c), (\bm{U},b)} = \tr(\bm{V}\bm{U}^T) + cb,
\end{equation}
whenever $(\bm{A},s) \in \Theta$, $(\bm{V},c)$, 
$(\bm{U},b) \in T_{(\bm{A},s)}\Theta \cong \mathbb{R}^{p \times q} \times \mathbb{R}$.
For later convenience,
let us define two functions $\Phi: \Theta \to \mathbb{R}^{p \times p}$
and $\bm{\Sigma}: M \to \mathbb{R}^{p \times p}$
by setting, for every $[\bm{A},s] \in M$:

\begin{equation}\label{eq:Sigma_Phi_def}
  \bm{\Sigma}([\bm{A},s]) := \Phi((\bm{A},s)) := \bm{A}\bm{A}^T + s\bm{I}_p.
\end{equation}
In other words, $\bm{\Sigma} \circ \pi = \Phi$
gives the covariance matrix.
\eqref{eq:original_model} is then reparametrized as

\begin{equation}\label{eq:quotient_model}
  \mathcal{P} = \curly{N_p\bracket{\bm{0}, \bm{\Sigma}([\bm{A},s])}}_{[\bm{A},s]\in M}
\end{equation}
and this is an identifiable model we shall work with.

\section{Basic Facts about $(M,g)$}

Working with Riemannian submersion, 
we shall as usual start with computations regarding the horizontal and verticle 
tangent spaces.

\begin{proposition}\label{thm:horizontal_verticle}
\,
\begin{enumerate}
  \item The map $\bm{\Sigma}$ defined in \eqref{eq:Sigma_Phi_def}
is a smooth injective immersion.
  \item At every $(\bm{A},s) \in \Theta$,
  the verticle tangent space 

  \begin{equation}\label{eq:verticle_space}
  V_{(\bm{A},s)} :=\ker(d\pi_{(\bm{A},s)}) = \ker(d\Phi_{(\bm{A},s)})
  \end{equation}
  equals to the collection of all tangent vectors 
of the form $(\bm{A} \bm{\Omega}, 0) \in T_{(\bm{A},s)}\Theta$,
with $\bm{\Omega}$ ranged over all $q \times q$ skew symmetric matrices.
  \item At every $(\bm{A},s) \in \Theta$,
  the horizontal tangent space $H_{(\bm{A},s)} := V_{(\bm{A},s)}^{\perp}$
equals to the collection of all tangent vectors 
of the form $(\bm{V}, c) \in T_{(\bm{A},s)}\Theta$
ranged over all $c \in \mathbb{R}$ and all $q \times q$ matrices $\bm{V}$
satisfying $\bm{V}^T \bm{A} = \bm{A}^T \bm{V}$.
\end{enumerate}
\end{proposition}

\begin{proof}
Fix a point $(\bm{A},s) \in \Theta$
throughout this proof.
Before we start the proof,
let us first study the set $\ker(d\Phi_{(\bm{A},s)})$,
where $\Phi$ is given in \eqref{eq:Sigma_Phi_def}.
For $(\bm{V}, c) \in T_{(\bm{A},s)}\Theta$, 
define a curve $\gamma : (-\epsilon, \epsilon) \to \Theta$
by setting $\gamma(t) = (\bm{A}+t\bm{V}, s + tc)$.
Then:

\begin{equation*}
  d \Phi_{(\bm{A},s)} ((\bm{V}, c)) = \frac{d}{dt} \bigg|_{t = 0} \Phi(\gamma(t)) = \bm{A}\bm{V}^T + \bm{V} \bm{A}^T + c\bm{I}_p.
\end{equation*}
If $(\bm{V}, c) \in \ker(d\Phi_{(\bm{A},s)})$,
then

\begin{equation}\label{eq:ker_Phi_equation}
  \bm{A}\bm{V}^T + \bm{V} \bm{A}^T + c\bm{I}_p = \bm{0}.
\end{equation}
If $\bm{u} \in \mathbb{R}^p$ is any nonzero vector orthogonal to
all column vectors of $\bm{A}$, then $\bm{A}^T \bm{u} = \bm{0}$ and

\begin{equation*}
  0 = \bm{u}^T (\bm{A}\bm{V}^T + \bm{V} \bm{A}^T + c\bm{I}_p) \bm{u}
  = c \, \bm{u}^T \bm{u}.
\end{equation*}
Given that $\bm{u} \neq \bm{0}$, we must have $c = 0$
and \eqref{eq:ker_Phi_equation} reduces to

\begin{equation}\label{eq:ker_Phi_equation_reduced}
  \bm{A}\bm{V}^T + \bm{V} \bm{A}^T = \bm{0}.
\end{equation}
Apply $\bm{u}$ to the right to get:

\begin{equation*}
  \bm{A}\bm{V}^T\bm{u} = \bm{0}.
\end{equation*}
Given that $\bm{A}$ has full rank, 
rank nullity theorem forces 
\begin{equation*}
  \bm{V}^T\bm{u} = \bm{0}.
\end{equation*}
Therefore, we have proven that every vector $\bm{u}$
orthgonal to columns vectors of $\bm{A}$
will also be orthgonal to column vectors of $\bm{V}$.
This means $\bm{V}=\bm{A} \bm{\Omega}$ for some 
$q \times q$ matrix $\bm{\Omega}$.
Substitute this back to \eqref{eq:ker_Phi_equation_reduced}
gives 

\begin{equation*}
  \bm{A}(\bm{\Omega}^T + \bm{\Omega})\bm{A}^T = \bm{0}.
\end{equation*}
Applying $(\bm{A}^T \bm{A})^{-1} \bm{A}^T$ 
to the left and $\bm{A}(\bm{A}^T \bm{A})^{-1}$ to the right 
yields $\bm{\Omega}^T + \bm{\Omega} = \bm{0}$.
Therefore, $\bm{\Omega}$ is skew-symmetric.
Conversely, \eqref{eq:ker_Phi_equation_reduced} 
is satisfied by $\bm{V}=\bm{A} \bm{\Omega}$
with any skew-symmetric matrix $\bm{\Omega}$.
We have proven that $\ker(d\Phi_{(\bm{A},s)})$
equals to the collection of all tangent vectors 
of the form $(\bm{A} \bm{\Omega}, 0) \in T_{(\bm{A},s)}\Theta$,
with $\bm{\Omega}$ ranged over all $q \times q$ skew symmetric matrices.

We now move back to prove 1 and 2 at the same time
by showing \eqref{eq:verticle_space}.
Clearly, $\bm{\Sigma}$ is well-defined 
and is injective.
The smoothness of $\bm{\Sigma}$
is guaranteed by relation $\bm{\Sigma} \circ \pi = \Phi$
(see for example \cite{Lee_2013} Theorem 4.29).
By the chain rule, we have 

\begin{equation} \label{eq:differential_Phi}
  d \Phi_{(\bm{A},s)} = d \bm{\Sigma}_{[\bm{A},s]} \circ d \pi_{(\bm{A},s)}.
\end{equation}
It follows immediately that 
$\ker(d\Phi_{(\bm{A},s)}) \subseteq \ker(d\pi_{(\bm{A},s)})$.
The other direction will follow if we can prove that 
$d \bm{\Sigma}_{[\bm{A},s]}$ is injective,
or equivalently that $\bm{\Sigma}$ is an immersion.

To do so, suppose that $\bm{v} \in T_{[\bm{A},s]}M$ is 
such that $d \bm{\Sigma}_{[\bm{A},s]} (\bm{v}) = 0$.
We prove that $\bm{v} = \bm{0}$.
Let $(\bm{V},c) \in T_{(\bm{A},s)}\Theta$ 
be the horizontal lift of $\bm{v}$ at $(\bm{A},s)$,
meaning that $(\bm{V},c)$ is the 
unique horizontal tangent vector at $(\bm{A},s)$
such that
 $\bm{v} = d \pi_{(\bm{A},s)}((\bm{V},c))$.
\eqref{eq:differential_Phi} then implies that 
$(\bm{V},c) \in \ker(d \Phi_{(\bm{A},s)})$
and the proceeding discussion shows that 
$c = 0$ and $\bm{V} = \bm{A} \bm{\Omega}$ 
for some skew-symmetric $\bm{\Omega}$.
To proceed, 
we look at the 
curve $\gamma: (-\epsilon, \epsilon) \to \Theta$
defined by $\gamma(t) = (\bm{A}e^{t \bm{\Omega}}, s)$.
Clearly $\gamma(0)=(\bm{A}, s)$ and $\gamma'(0) = (\bm{A} \bm{\Omega}, 0)$.
Moreover, the matrix $e^{t \bm{\Omega}}$ is always orthogonal 
for any $t$ and any skew-symmetric $\bm{\Omega}$ 
so that $\pi(\gamma(t)) = [\bm{A},s]$ for all $t$.
We have 

\begin{equation*}
  \bm{v} =
  d \pi_{(\bm{A},s)}((\bm{A} \bm{\Omega}, 0)) = 
  d \pi_{(\bm{A},s)} (\gamma'(0)) = \frac{d}{dt} \bigg|_{t=0} \pi(\gamma(t)) = \bm{0} \in T_{[\bm{A},s]}M.
\end{equation*}
Therefore, $d \bm{\Sigma}_{[\bm{A},s]}$ is injective and
the verticle tangent space is the same as $\ker(d\Phi_{(\bm{A},s)})$.

To compute the horizontal tangent space, any $(\bm{V}, c) \in V_{(\bm{A},s)}^{\perp}$ 
will satisfy 

\begin{equation}\label{eq:orthogonal_horizontal}
  0 = g_{(\bm{A},s)} \bracket{(\bm{V}, c), (\bm{A} \bm{\Omega}, 0)} = \tr(\bm{A}^T \bm{V} \bm{\Omega})
\end{equation}
for all skew-symmetric $\bm{\Omega}$.
Decomposing $\bm{A}^T \bm{V} = \bm{S} + \bm{K}$, 
where $\bm{S}$ is the symmetric part and $\bm{K}$ is the skew-symmetric part,
we find out that 

\begin{equation*}
  0 = \tr(\bm{S} \bm{\Omega}) + \tr(\bm{K} \bm{\Omega}) = 0 + \tr(\bm{K} \bm{\Omega}) = \tr(\bm{K} \bm{\Omega}).
\end{equation*}
Setting $\bm{\Omega} = \bm{K}^T$ will yield that $\bm{K} = \bm{0}$.
Therefore, $\bm{A}^T \bm{V}$ is a symmetric matrix.
Conversely, any $\bm{V}$ such that $\bm{A}^T \bm{V}$ is symmetric will satisfy 
\eqref{eq:orthogonal_horizontal}. 
The proof is complete.
\end{proof}

\begin{proposition}\label{thm:exponential_map}
Given $[\bm{A},s] \in M$ and $\bm{v} \in T_{[\bm{A},s]}M$, 
geodesics in $M$ starting at $[\bm{A},s]$ with initial velocity 
$\bm{v}$ takes the format of curves 
$\gamma: I \to M$, where $I$ is an open interval containing $0$,
defined by $\gamma(t) = [\bm{A} + t\bm{V}, s + tc]$,
where $(\bm{V},c) \in H_{(\bm{A},s)}$ is the horizontal lift 
of $\bm{v}$ at $ (\bm{A},s)$.
Consequently, the exponential map 
is computed as $\text{Exp}_{[\bm{A},s]}(\bm{v}) = [\bm{A} + \bm{V}, s + c]$,
at least well-defined over an open ball $\mathcal{E}_{[\bm{A},s]}$
of $\bm{0} \in T_{[\bm{A},s]}M \cong \mathbb{R}^{pq+1-\frac{q(q-1)}{2}}$ 
of radius $\min(\sigma_{\text{min}}(\bm{A}),s)$,
where $\sigma_{\text{min}}(\bm{A})$ denotes the smallest singular value of $\bm{A}$.

all $\bm{v}$ such that 
$\sqrt{g_{[\bm{A},s]}(\bm{v},\bm{v})}=\abs{\bm{v}} < \min(\sigma_{\text{min}}(\bm{A}),s)$,
where $\sigma_{\text{min}}(\bm{A})$ is the smallest singular value of $\bm{A}$.
\end{proposition}

\begin{proof}
First of all, let us quickly check that the curve $\gamma$ is well defined.
Indeed, if $(\bm{A}\bm{Q},s)$, where $\bm{Q} \in O(q)$, is another representative of $[\bm{A},s]$,
let $(\bm{V}',c')$ be the horizontal lift of $\bm{v}$ to $(\bm{A}\bm{Q},s)$.
With the function $R_{\bm{Q}}: \Theta \to \Theta$ defined by $R_{\bm{Q}}((\bm{A},s)) = (\bm{A}\bm{Q},s)$,
we have 

\begin{equation*}
  \bm{v} = d\pi_{(\bm{A},s)} (\bm{V},c) = d(\pi \circ R_{\bm{Q}})_{(\bm{A},s)} (\bm{V},c)
  = d \pi_{(\bm{A}\bm{Q},s)} (dR_{\bm{Q}})_{(\bm{A},s)}(\bm{V},c)
  = d \pi_{(\bm{A}\bm{Q},s)} (\bm{V}\bm{Q},c).
\end{equation*}
Thus, $\bm{V}' = \bm{V}\bm{Q}$ and $c'=c$.
Substitute this into the definition of $\gamma$ 
will yield the exact same curve.

A well-known property of 
geodesics in case of Riemannian submersion (see for example 
\cite{O_neill_1967} Section 2 Corollary 2 or \cite{Valencia_2021} Theorem 3.6)
is that any geodesic $c$ in $\Theta$ with initial velocity being horizontal 
will always have horizontal velocities and 
$\gamma = \pi \circ c$
will remain a geodesic in $M$.
Therefore, $\text{Exp}_{[\bm{A},s]}(\bm{v}) = [\bm{A} + \bm{V}, s + c]$
is well defined for all vectors $\bm{v} \in T_{[\bm{A},s]}M$
of small enough norm
making sure that $(\bm{A} + \bm{V}, s + c) \in \Theta$.
To have a quantified upper bound of the norm,
we cite
Weyl's perturbation inequality (Chapter 7.3 Problem 16 equation (7.3.15) of \cite{Horn_Johnson_2013})
to get

\begin{equation*}
  \sigma_{\text{min}}(\bm{A} + \bm{V}) \geqslant \sigma_{\text{min}}(\bm{A}) - \sigma_{\text{max}}(\bm{V})
  \geqslant \sigma_{\text{min}}(\bm{A}) - \abs{\bm{v}} > 0,
\end{equation*}
whenever $\abs{\bm{v}} < \min(\sigma_{\text{min}}(\bm{A}),s)$.
Here $\sigma_{\text{max}}(\bm{V})$ denotes the largest singular value of $\bm{V}$,
who is smaller than the Euclidean norm of $\bm{V}$ and hence norm of $\bm{v}$.
Thus, for all $\bm{v} \in \mathcal{E}_{[\bm{A},s]}$, $\rank(\bm{A} + \bm{V}) = q$
and the exponential map is well-defined.
\end{proof}

To compute the logarithmic map, we need first preparatory lemmas that 
work out an orthogonal matrix based on given points in $M$.

\begin{lemma}\label{thm:q_classification}
If $\bm{L}$ is an invertible $q \times q$ matrix
and $\bm{Q} \in O(q)$ are such that $\bm{L}\bm{Q}$ is symmetric,
then there exists matrices $\bm{R}\in O(q)$, $\bm{P} \in O(q)$, 
a $q \times q$ diagonal matrix $\bm{D}$ whose diagonal entries are $1$ or $-1$,
and a $q \times q$ diagonal matrix $\bm{S}$ containing the singular values 
of $\bm{L}$ in increasing order 
such that $\bm{L} = \bm{R}\bm{S}\bm{P}^T$
and $\bm{Q} = \bm{P}\bm{D}\bm{R}^T$.
\end{lemma}

\begin{proof}
Let us start by applying the singular value decomposition 
of $\bm{L}$ to get matrices $\bm{R}'\in O(q)$, $\bm{P}' \in O(q)$
and $\bm{S}$ such that $\bm{L} = \bm{R}'\bm{S}(\bm{P}')^T$.
Substituting the given condition that $\bm{L}\bm{Q}$ is symmetric will yield that
$\bm{S}\bm{J} = \bm{J}^T\bm{S}$, where $\bm{J} := (\bm{P}')^T \bm{Q} \bm{R}'$.

Clearly $\bm{J} \in O(q)$.
The equation 

\begin{equation*}
  \bm{S}^2 = \bm{S}\bm{J}\bm{J}^T\bm{S} = (\bm{S}\bm{J})(\bm{S}\bm{J})^T 
  = (\bm{J}^T\bm{S})(\bm{J}^T\bm{S})^T = \bm{J}^T\bm{S}^2\bm{J}
\end{equation*}
shows that 
\begin{equation*}
  \bm{J}\bm{S}^2=\bm{S}^2\bm{J}.
\end{equation*}
Looking at each entry of both size,
we end up with 
\begin{equation*}
  \bm{J}_{ij}(\bm{S}_{jj}^2-\bm{S}_{ii}^2) = 0
\end{equation*}
for $i,j = 1, \ldots q$.
Given that $\bm{L}$ is invertible, its singular values are all strictly positive.
Therefore, in fact we have 
\begin{equation}\label{eq:j_s_equation}
  \bm{J}_{ij}(\bm{S}_{jj}-\bm{S}_{ii}) = 0
\end{equation}
for all $i$ and all $j$ so that $\bm{J}\bm{S}=\bm{S}\bm{J}=\bm{J}^T\bm{S}$.
Invert $\bm{S}$ yields that $\bm{J}$ is actually symmetric.
Being a symmetric orthogonal matrix, $\bm{J}^2 = \bm{I}_q$ and the eigenvalues of $\bm{J}$ 
must be $1$ or $-1$.
Another look at \eqref{eq:j_s_equation} shows that $\bm{J}$ is 
a block-diagonal matrix, diagonal if and only if
all diagonal entries of $\bm{S}$ are distinct.
Furthermore, each block of $\bm{J}$ must still be orthogonal and symmetric.

We now diagonalize $\bm{J}$ based on the above information.
Suppose that $\bm{S} = \oplus_{i=1}^k \bm{s}_i\bm{I}_{m_i}$,
where $\bm{s}_1 > \ldots > \bm{s}_k$ are the unique diagonal entries of 
$\bm{S}$ and each repeats for $m_i$ times.
Thus, $\bm{J}$ has $k$ blocks.
Write the $i$th block by $\bm{J}_i$ so that
$\bm{J} = \oplus_{i=1}^k \bm{J}_i$.
The spectral theorem then allows us to orthogonally diagonalize 
$\bm{J}_i = \bm{K}_i\bm{D}_i\bm{K}_i^T$,
where 
$\bm{K}_i \in O(m_i)$ and
$\bm{D}_i$ is an $m_i \times m_i$ diagonal matrix with $1$ or $-1$ diagonal 
entry. Put $\bm{K} = \oplus_{i=1}^k \bm{K}_i$
and $\bm{D} = \oplus_{i=1}^k \bm{D}_i$.
Then $\bm{K} \in O(q)$, $\bm{D}$ is a $q \times q$ diagonal matrix 
with $1$ or $-1$ diagonal entry, and $\bm{J} = \bm{K}\bm{D}\bm{K}^T$.
This involved construction leads us to the outcome that 
$\bm{K}$ in fact commutes with $\bm{S}$:

\begin{equation*}
  \bm{K}\bm{S} = \bigoplus_{i=1}^k \bm{s}_i\bm{K}_i = \bm{S}\bm{K}.
\end{equation*}
Put $\bm{P} = \bm{P}'\bm{K}$,
$\bm{R} = \bm{R}'\bm{K}$.
We have 
\begin{equation*}
  \bm{Q} = \bm{P}' \bm{J} (\bm{R}')^T = \bm{P}' \bm{K}\bm{D}\bm{K}^T (\bm{R}')^T
  = \bm{P} \bm{D} \bm{R}^T
\end{equation*}
and
\begin{equation*}
  \bm{R}\bm{S}\bm{P}^T = 
  \bm{R}'\bm{K} \bm{S}\bm{K}^T(\bm{P}')^T
  = \bm{R}'\bm{S} \bm{K}\bm{K}^T(\bm{P}')^T
  = \bm{R}'\bm{S} (\bm{P}')^T =\bm{L}.
\end{equation*}
This completes the proof.
\end{proof}

\begin{lemma}\label{thm:singular_PR}
If $\bm{L}$ is an invertible $q \times q$ matrix
with singular value decomposition $\bm{L} = \bm{R}\bm{S}\bm{P}^T$,
then 

\begin{equation*}
  \bm{P}\bm{R}^T = (\bm{L}^T \bm{L})^{-\frac{1}{2}}\bm{L}^T.
\end{equation*}
This means, although neither $\bm{P}$ nor $\bm{R}$ is uniquely determined,
$\bm{P}\bm{R}^T$ is uniquely determined.
\end{lemma}
\begin{proof}
Given that

\begin{equation*}
  \bm{L}^T\bm{L} = \bm{P}\bm{S}^2\bm{P}^T,
\end{equation*}
we have
\begin{equation*}
  (\bm{L}^T \bm{L})^{\frac{1}{2}} = \bm{P}\bm{S}\bm{P}^T.
\end{equation*}
Therefore:
\begin{equation*}
  \bm{L} = \bm{R}\bm{P}^T\bm{P}\bm{S}\bm{P}^T = \bm{R}\bm{P}^T (\bm{L}^T \bm{L})^{\frac{1}{2}}.
\end{equation*}
The result follows from inverting $(\bm{L}^T \bm{L})^{\frac{1}{2}}$ and take transpose.
\end{proof}

\begin{proposition}\label{thm:logarithmic_map}
For every $[\bm{A},s] \in M$ there exists an open
neighborhood $U \subset M$ containing $[\bm{A},s]$ such that

\begin{enumerate}
  \item For every $[\bm{B},t] \in U$, $\bm{A}^T\bm{B}$ is invertible.
  \item The logarithmic map $\text{Exp}^{-1}_{[\bm{A},s]}:U \to T_{[\bm{A},s]}M$ 
  is well-defined and is smooth in $U$. In fact:

  \begin{equation*}
    \text{Exp}^{-1}_{[\bm{A},s]}([\bm{B},t]) 
    = d \pi_{(\bm{A},s)}((\bm{B}\bm{Q}(\bm{A},\bm{B})-\bm{A}, t-s)),
  \end{equation*}
  where 

  \begin{equation*}
    \bm{Q}(\bm{A},\bm{B}) 
    = (\bm{B}^T \bm{A} \bm{A}^T \bm{B})^{- \frac{1}{2}} \bm{B}^T \bm{A} \in O(q).
  \end{equation*}
\end{enumerate}
\end{proposition}

\begin{proof}
First of all, 
if $\bm{R} \in O(q)$ and $\bm{P} \in O(q)$,
then $\bm{Q}(\bm{A}\bm{P},\bm{B}\bm{R}) = \bm{R}^T\bm{Q}(\bm{A},\bm{B})\bm{P}$
and the same computation as in the proof of 
Proposition~\ref{thm:exponential_map}
shows that the formula of logarithmic map is indeed well-defined.

Given that the determinant is a continuous function
and $\det(\bm{A}^T\bm{A}) > 0$,
there exists a neighborhood $U_{\Theta} \subset \Theta$ 
containing $(\bm{A},s)$ 
such that $\det(\bm{A}^T\bm{B}) > 0$ 
whenever $(\bm{B},t) \in U_{\Theta}$.
Thus,
for any $[\bm{B},t] \in U_1 := \pi(U_{\Theta}) \subset M$,
$\bm{A}^T\bm{B}$ is invertible.

Next, 
it is a well-known fact that $\text{Exp}_{[\bm{A},s]}$ 
is a local diffeomorphism 
and thus there exists an open neighborhood $U_2 \subset M$ containing $[\bm{A},s]$
such that the logarithmic map $\text{Exp}^{-1}_{[\bm{A},s]}$
is well-defined and is smooth in $U_2$.

We now compute $\text{Exp}^{-1}_{[\bm{A},s]}$
for $[\bm{B},t] \in U := U_1 \cap U_2$.
Suppose that $\bm{v} \in T_{[\bm{A},s]}M$ 
satisfies $\bm{v} = \text{Exp}^{-1}_{[\bm{A},s]}([\bm{B},t])$.
By Proposition~\ref{thm:exponential_map},
$[\bm{B},t] = [\bm{A} + \bm{V},s+c]$,
where $(\bm{V}, c)$ is the horizontal lift of $\bm{v}$ to $(\bm{A}, s)$.
It follows that $c = t-s$ and that
there exists a $\bm{Q} \in O(q)$
such that $\bm{V} = \bm{B}\bm{Q}-A$.
Or equivalently:

\begin{equation}\label{eq:logarithmic_general_in_proof}
  \text{Exp}^{-1}_{[\bm{A},s]}([\bm{B},t]) 
    = d \pi_{(\bm{A},s)}((\bm{B}\bm{Q}-\bm{A}, t-s)).
\end{equation}
Given that $(\bm{V}, c)$ is a horizontal tangent vector, 
Proposition~\ref{thm:horizontal_verticle} shows that 

\begin{equation*}
  \bm{A}^T(\bm{B}\bm{Q}-\bm{A}) = (\bm{B}\bm{Q}-\bm{A})^T \bm{A}
\end{equation*}
and consequently $\bm{A}^T \bm{B}\bm{Q}$ is symmetric.
Lemma~\ref{thm:q_classification} with $\bm{L} = \bm{A}^T \bm{B}$
shows the existence of $\bm{R}$, $\bm{P} \in O(q)$,
a diagonal matrix $\bm{S}$ with the singular values of $\bm{A}^T \bm{B}$
in increasing order, and a $q \times q$ diagonal matrix $\bm{D}$
with diagonal entries $1$ or $-1$
such that $\bm{A}^T \bm{B} = \bm{R}\bm{S}\bm{P}^T$
and $\bm{Q} = \bm{P}\bm{D}\bm{R}^T$.
\eqref{eq:logarithmic_general_in_proof} now becomes
\begin{equation}\label{eq:logarithmic_PR_in_proof}
  \text{Exp}^{-1}_{[\bm{A},s]}([\bm{B},t]) 
    = d \pi_{(\bm{A},s)}((\bm{B}\bm{P}\bm{D}\bm{R}^T-\bm{A}, t-s)).
\end{equation}
Now let $\bm{B} \to \bm{A}$ and $t \to s$.
The left hand size of \eqref{eq:logarithmic_PR_in_proof}
now becomes $\text{Exp}^{-1}_{[\bm{A},s]}([\bm{A},s]) = \bm{0} \in T_{[\bm{A},s]}M$.
The singular value decomposition converges to 
the orthogonal diagonalization because $\bm{A}^T \bm{A}$
is symmetric, meaning that $\bm{P} =  \bm{R}$. Thus

\begin{equation*}
  \bm{A}\bm{P}\bm{D}\bm{P}^T-\bm{A} = \bm{0}.
\end{equation*}
Therefore, $\bm{P}\bm{D}\bm{P}^T = \bm{I}_q$
and $\bm{D} = \bm{I}_q$.
Lemma~\ref{thm:singular_PR} then shows 
that 
\begin{equation*}
  \bm{Q} = \bm{P}\bm{R}^T = (\bm{B}^T \bm{A} \bm{A}^T \bm{B})^{- \frac{1}{2}} \bm{B}^T \bm{A}.
\end{equation*}
This completes the proof.
\end{proof}

\begin{remark}
Unfortunately, the logarithmic map is not globally defined
in the standard way
even over all $[\bm{B},t]$ such that $\bm{A}^T\bm{B}$ is invertible.
For a counterexample, take $p = 2$, $q = 1$,
$\bm{A} = [1,0]^T$, $\bm{B} = [1,1]^T$ and $s = t = 1$.
In this case, $O(1) = \{1,-1\}$. Then, 
although $\bm{A}^T\bm{B} = 1$ is invertible,
we still have 

\begin{equation*}
  \text{Exp}_{[\bm{A},s]}\bracket{d\pi_{(\bm{A},s)}((\bm{B}-\bm{A},0))}
  = [\bm{B},1] = [-\bm{B},1]
  = \text{Exp}_{[\bm{A},s]}\bracket{d\pi_{(\bm{A},s)}((-\bm{B}-\bm{A},0))}.
\end{equation*}
The reason is that,
even when we assume $\bm{A}^T \bm{B}$ is invertible, 
deducing $\bm{D} = \bm{I}_q$ 
relies on the local smoothness of the logarithmic map and the 
procedure of letting $[\bm{B},t] \to [\bm{A},s]$.
If $[\bm{B},t]$ is far away from $[\bm{A},s]$,
then $\bm{Q} = \bm{P}\bm{D}\bm{R}^T$ with some $\bm{D} \neq \bm{I}_q$
can still produce a valid horizontal vector whose image under
$\text{Exp}_{[\bm{A},s]}$ is $[\bm{B},t]$,
so that $\text{Exp}_{[\bm{A},s]}$ fails to be injective
and admits no globally defined inverse.

Nevertheless, the choice $\bm{D} = \bm{I}_q$ has an interesting 
property of selecting the geodesic of minimal speed
and length when $[\bm{B},t]$ is any point 
such that $\bm{A}^T \bm{B}$ is invertible.
To see this, 
consider a general candidate $\bm{Q}' = \bm{P}\bm{D}\bm{R}^T$ and
write $\bm{v'} = d\pi_{(\bm{A},s)}((\bm{B}\bm{Q}'-\bm{A}, t-s))$,
so that $(\bm{B}\bm{Q}'-\bm{A}, t-s)$ is the horizontal lift of $\bm{v'}$
at $(\bm{A},s)$ and $\text{Exp}_{[\bm{A},s]}(\bm{v'}) = [\bm{B},t]$.
Let us compute the squared norm of $\bm{v'}$:

\begin{align*}
  & \tab[0.4cm] g_{[\bm{A},s]}(\bm{v'}, \bm{v'})
  = g_{(\bm{A},s)}^{\Theta}\bracket{(\bm{B}\bm{Q}'-\bm{A}, t-s), (\bm{B}\bm{Q}'-\bm{A}, t-s)} \\
  &= \tr\bracket{(\bm{B}\bm{Q}'-\bm{A})(\bm{B}\bm{Q}'-\bm{A})^T} + (t-s)^2 
  = \tr(\bm{B}\bm{B}^T) + \tr(\bm{A}\bm{A}^T) - 2\tr(\bm{S}\bm{D}) + (t-s)^2,
\end{align*}
where the last equality substituted
$\bm{A}^T\bm{B}\bm{Q}' = \bm{R}\bm{S}\bm{P}^T\bm{P}\bm{D}\bm{R}^T = \bm{R}\bm{S}\bm{D}\bm{R}^T$
and used the cyclic property of the trace.
The fact that
the diagonal entries of $\bm{S}$ are strictly positive
show that
$g_{[\bm{A},s]}(\bm{v'}, \bm{v'})$ is minimized precisely when $\bm{D} = \bm{I}_q$.
Given that the geodesic starting at $[\bm{A},s]$
with initial velocity $\bm{v'}$ reaches $[\bm{B},t]$ at time $1$ with constant speed
$\sqrt{g_{[\bm{A},s]}(\bm{v'}, \bm{v'})}$, its length is $\sqrt{g_{[\bm{A},s]}(\bm{v'}, \bm{v'})}$.
The choice $\bm{D} = \bm{I}_q$ gives the slowest and hence shortest geodesic.
To illustrate,
let us go back to the counterexample in the previous paragraph.
Put $\bm{v}_{\pm} := d\pi_{(\bm{A},s)}((\pm\bm{B}-\bm{A},0))$.
Direct computation shows $g_{[\bm{A},s]}(\bm{v}_+,\bm{v}_+) = 1$ and $g_{[\bm{A},s]}(\bm{v}_-,\bm{v}_-) = 5$
so that the option corresponding to $\bm{D} = \bm{I}_1=1$
gives the slowest speed.
Therefore, it is actually possible to define 
$\text{Exp}^{-1}_{[\bm{A},s]}$
over all $[\bm{B},t]$ such that $\bm{A}^T \bm{B}$ is invertible
by manually asking it to output the minimal speed candidate 
and this definition will yield the same $\bm{Q}(\bm{A},\bm{B})$
formula. 

However, the situation totally breaks down 
in case $\bm{A}^T \bm{B}$ is not invertible.
Consider again $p =2$, $q=1$, $\bm{A} = [1,0]^T$, $s = t = 1$,
but this time $\bm{B} = [0,1]^T$.
Then $g_{[\bm{A},s]}(\bm{v}_+,\bm{v}_+) = g_{[\bm{A},s]}(\bm{v}_-,\bm{v}_-) = 2$
and there are two distinct geodesics of the same speed 
so $\text{Exp}^{-1}_{[\bm{A},s]}([\bm{B},t])$
cannot be well-defined.

(Picture: there is a hole at $[0,0]^T$. 
Starting $\bm{A}= [1,0]^T$, as $\bm{B}= [0,1]^T$ is glued to $[0,-1]^2$,
$\bm{v}_+$ points from $[1,0]^T$ to $[0,1]^T$,
$\bm{v}_-$ points from $[1,0]^T$ to $[0,-1]^T$)

Given the above situation and (the expectation that) 
our work should not require a global definition,
we stick to the standard definition of logarithmic map
only defined locally around $[\bm{A},s]$.
\end{remark}

\section{Properties of the Probabilistic Principal Component Analysis Model}
We now show that the statistical model $\mathcal{P}$ 
given by \eqref{eq:quotient_model}
is \textit{differentiable in quadratic mean}
and has \textit{positive definite Fisher information operator} 
in the sense developed in \cite{Sun_Lin_Liu_2026}, definition 
III.1 and III.2.

Let us introduce a few notations.
For each $[\bm{A},s] \in M$,
let

\begin{equation}\label{eq:p_def}
  p(\bm{x}; [\bm{A},s])
  = (2\pi)^{-\frac{p}{2}} \text{det}^{-\frac{1}{2}}(\bm{\Sigma}([\bm{A},s])) \exp \bracket{- 
  \frac{1}{2}\bm{x}^T(\bm{\Sigma}([\bm{A},s]))^{-1}\bm{x}}
\end{equation}
be the density of $N_p(\bm{0}, \bm{\Sigma}([\bm{A},s]))$.
Define two maps $l_{[\bm{A},s]}$ and $r_{[\bm{A},s]}$
with domain $\mathcal{E}_{[\bm{A},s]} \subset T_{[\bm{A},s]}M \cong \mathbb{R}^{pq+1-\frac{q(q-1)}{2}}$
and codomain $L^2(\mathbb{R}^p)$ by setting,
for all $\bm{v} \in T_{[\bm{A},s]}M$,
\begin{equation}\label{eq:l_def}
  l_{[\bm{A},s]}(\bm{v})(\bm{x}) = \log \bracket{p(\bm{x}; \text{Exp}_{[\bm{A},s]}(\bm{v}))}
\end{equation}
and
\begin{equation}\label{eq:r_def}
  r_{[\bm{A},s]}(\bm{v})(\bm{x}) = \sqrt{p(\bm{x}; \text{Exp}_{[\bm{A},s]}(\bm{v}))}.
\end{equation}

\begin{theorem}
$r_{[\bm{A},s]}$ is Fréchet differentiable at any $\bm{v} \in \mathcal{E}_{[\bm{A},s]}$.
In particular, $\mathcal{P}$ is differentiable in quadratic mean with score 
\begin{equation}\label{eq:score}
  s(\bm{x}; [\bm{A},s])(\bm{h})
  = \tr \curly{\bracket{\bm{\Sigma}([\bm{A},s])}^{-1}
  \squared{\bm{x}\bm{x}^T - \bracket{\bm{\Sigma}([\bm{A},s])}}
  \bracket{\bm{\Sigma}([\bm{A},s])}^{-1} \bracket{D_{\bm{h}}\bm{\Sigma}([\bm{A},s])}},
\end{equation}
where 
\begin{equation}\label{eq:Sigma_directional}
  D_{\bm{h}}\bm{\Sigma}([\bm{A},s])
  := \frac{d}{dt}\bigg|_{t=0} \bm{\Sigma}(\text{Exp}_{[\bm{A},s]}(t\bm{h}))
  = 2\bm{U}\bm{A}^T + b  \bm{I}_p,
\end{equation}
if $\bm{h}\in T_{[\bm{A},s]}M$ lifts to $(\bm{U},b)$ at $(\bm{A},s)$.
Furthermore, $\mathcal{P}$ is a regular model in the sense that 
the Fisher information operator 
\begin{equation}\label{eq:G_def}
  G_{[\bm{A},s]} (\bm{h}_1,\bm{h}_2)
  = \tr \curly{\bracket{\bm{\Sigma}([\bm{A},s])}^{-1}
  D_{\bm{h}_1}\bm{\Sigma}([\bm{A},s])
  \bracket{\bm{\Sigma}([\bm{A},s])}^{-1}
  D_{\bm{h}_2}\bm{\Sigma}([\bm{A},s])}
\end{equation}
 is positive definite.
\end{theorem}

\begin{proof}
The proof for the Fréchet differentiability
is long but routine.
We compute the Gâteaux derivative
of $r_{[\bm{A},s]}$ at $\bm{v} \in \mathcal{E}_{[\bm{A},s]}$
in a given direction $\bm{h} \in T_{[\bm{A},s]}M$
and show that the Gâteaux derivative is the Fréchet derivative 
by a direct analysis of the remainder term.
At the very end, we specialize our computations to 
$\bm{v} = \bm{0}$ to prove \eqref{eq:score}
and \eqref{eq:G_def}.

Given $\bm{v}$ and $\bm{h}$,
there exists an $\epsilon > 0$ such that
$\bm{v} + t\bm{h} \in \mathcal{E}_{[\bm{A},s]}$
whenever $t \in (-\epsilon, \epsilon)$.
If they respectively lift to $(\bm{V},c)$
and $(\bm{U},b)$ at $(\bm{A},s)$,
then $\bm{v} + t\bm{h}$ lifts to $(\bm{V} + t\bm{U},c+tb)$
at $(\bm{A},s)$.
First of all:

\begin{equation}\label{eq:Sigma_derivative}
  \begin{split}
  D_{\bm{h}}\bm{\Sigma}(\text{Exp}_{[\bm{A},s]}(\bm{v})) & :=
  \frac{d}{dt}\bigg|_{t=0} \bm{\Sigma}(\text{Exp}_{[\bm{A},s]}(\bm{v} + t\bm{h}))
  = \frac{d}{dt}\bigg|_{t=0} \curly{(\bm{A}+\bm{V} + t\bm{U})(\bm{A}+\bm{V} + t\bm{U})^T + (s+c+tb) \bm{I}_p}
  \\ & = \bm{U}\bm{V}^T + \bm{V}\bm{U}^T + 2\bm{U}\bm{A}^T + b  \bm{I}_p,
  \end{split}
\end{equation}
proving \eqref{eq:Sigma_directional} with special case $\bm{v}=\bm{0}$.
Secondly, for all $\bm{x} \in \mathbb{R}^p$:
\begin{equation}\label{eq:l_derivative}
  \begin{split}
  & \tab[0.4cm] D_{\bm{h}}l_{[\bm{A},s]}(\bm{v})(\bm{x}) :=
  \frac{d}{dt}\bigg|_{t=0} l_{[\bm{A},s]}(\bm{v} + t\bm{h})(\bm{x})
  \\ & = \frac{d}{dt}\bigg|_{t=0} \curly{-\frac{p}{2}\log(2\pi) 
  -\frac{1}{2}\log \det\bracket{\bm{\Sigma}(\text{Exp}_{[\bm{A},s]}(\bm{v} + t\bm{h}))}
  - \frac{1}{2}\bm{x}^T\bracket{\bm{\Sigma}(\text{Exp}_{[\bm{A},s]}(\bm{v} + t\bm{h}))}^{-1}\bm{x} }
  \\ & =
  -\frac{1}{2}\tr 
  \squared{\bracket{\bm{\Sigma}(\text{Exp}_{[\bm{A},s]}(\bm{v}))}^{-1}
  \bracket{\frac{d}{dt}\bigg|_{t=0}\bm{\Sigma}(\text{Exp}_{[\bm{A},s]}(\bm{v} + t\bm{h})) }}
  \\ & + \frac{1}{2}\bm{x}^T\bracket{\bm{\Sigma}(\text{Exp}_{[\bm{A},s]}(\bm{v}))}^{-1}
  \bracket{\frac{d}{dt}\bigg|_{t=0}\bm{\Sigma}(\text{Exp}_{[\bm{A},s]}(\bm{v} + t\bm{h}))}
  \bracket{\bm{\Sigma}(\text{Exp}_{[\bm{A},s]}(\bm{v}))}^{-1}\bm{x}
  \\ & =
  -\frac{1}{2} \tr 
  \squared{\bracket{\bm{\Sigma}(\text{Exp}_{[\bm{A},s]}(\bm{v}))}^{-1}
  \bracket{D_{\bm{h}}\bm{\Sigma}(\text{Exp}_{[\bm{A},s]}(\bm{v})) }}
  \\ & + \frac{1}{2}\bm{x}^T\bracket{\bm{\Sigma}(\text{Exp}_{[\bm{A},s]}(\bm{v}))}^{-1}
  \bracket{D_{\bm{h}}\bm{\Sigma}(\text{Exp}_{[\bm{A},s]}(\bm{v}))}
  \bracket{\bm{\Sigma}(\text{Exp}_{[\bm{A},s]}(\bm{v}))}^{-1}\bm{x}
  \\ & = \frac{1}{2} \tr \curly{\bracket{\bm{\Sigma}(\text{Exp}_{[\bm{A},s]}(\bm{v}))}^{-1}
  \squared{\bm{x}\bm{x}^T - \bracket{\bm{\Sigma}(\text{Exp}_{[\bm{A},s]}(\bm{v}))}}
  \bracket{\bm{\Sigma}(\text{Exp}_{[\bm{A},s]}(\bm{v}))}^{-1} \bracket{D_{\bm{h}}\bm{\Sigma}(\text{Exp}_{[\bm{A},s]}(\bm{v}))}}.
  \end{split}
\end{equation}
Finally:
\begin{equation}\label{eq:r_derivative}
  D_{\bm{h}}r_{[\bm{A},s]}(\bm{v})(\bm{x})
  = D_{\bm{h}} \exp \bracket{\frac{1}{2}l_{[\bm{A},s]}(\bm{v})(\bm{x})}
  = \frac{1}{2} \exp \bracket{\frac{1}{2}l_{[\bm{A},s]}(\bm{v})(\bm{x})} D_{\bm{h}}l_{[\bm{A},s]}(\bm{v})(\bm{x})
  = \frac{1}{2} r_{[\bm{A},s]}(\bm{v})(\bm{x}) D_{\bm{h}}l_{[\bm{A},s]}(\bm{v})(\bm{x}).
\end{equation}
To prove that \eqref{eq:r_derivative}
indeed defines a valid Gâteaux derivative,
we must prove that $D_{\bm{h}}r_{[\bm{A},s]}(\bm{v}) \in L^2(\mathbb{R}^p)$.
Luckily, the expectation and covariance of normal 
quadratic forms have already been very well studied in 
\cite{Gupta_Nagar_2000}.
For $\bm{X} \sim N_p(\bm{0}, \bm{\Sigma})$,
Theorem 7.7.1(i) 
states that 
\begin{equation*}
  \mathbb{E} \squared{\bm{X}^T \bm{Z} \bm{X}}
  = \tr \bracket{\bm{Z}\bm{\Sigma}}
\end{equation*}
for any $p \times p$ matrix $\bm{Z}$
and 
Theorem 7.7.2 states that 
\begin{equation*}
  \text{Cov} \bracket{\bm{X}^T \bm{Z} \bm{X}, \bm{X}^T \bm{Y} \bm{X}}
  = \tr(\bm{Z}\bm{\Sigma}\bm{Y}^T\bm{\Sigma}) + \tr(\bm{Z}^T\bm{\Sigma}\bm{Y}^T\bm{\Sigma})
\end{equation*} 
for any two $p \times p$ matrices $\bm{Y}$ and $\bm{Z}$.
Therefore, 
we can confirm without effort that the expectation of derivative of log density is zero:
\begin{equation*}
  \begin{split}
  & \tab[0.4cm] \int_{\mathbb{R}^p} D_{\bm{h}}l_{[\bm{A},s]}(\bm{v})(\bm{x})
  p(\bm{x}; \bm{\Sigma}(\text{Exp}_{[\bm{A},s]}(\bm{v})))\mathrm{d} \bm{x}
   \\ & = -\frac{1}{2} \tr 
  \squared{\bracket{\bm{\Sigma}(\text{Exp}_{[\bm{A},s]}(\bm{v}))}^{-1}
  \bracket{D_{\bm{h}}\bm{\Sigma}(\text{Exp}_{[\bm{A},s]}(\bm{v})) }}
  \\ & + \frac{1}{2}\int_{\mathbb{R}^p}\squared{\bm{x}^T\bracket{\bm{\Sigma}(\text{Exp}_{[\bm{A},s]}(\bm{v}))}^{-1}
  \bracket{D_{\bm{h}}\bm{\Sigma}(\text{Exp}_{[\bm{A},s]}(\bm{v}))}
  \bracket{\bm{\Sigma}(\text{Exp}_{[\bm{A},s]}(\bm{v}))}^{-1}\bm{x}} 
  p(\bm{x}; \bm{\Sigma}(\text{Exp}_{[\bm{A},s]}(\bm{v})))\mathrm{d} \bm{x} = 0
  \end{split}
\end{equation*} 
and, whenever $\bm{h}_1$ and $\bm{h}_2 \in T_{[\bm{A},s]}M$:
\begin{equation}\label{eq:fisher_computation}
  \begin{split}
  & \tab[0.4cm] 	
   \int_{\mathbb{R}^p} D_{\bm{h}_1}l_{[\bm{A},s]}(\bm{v})(\bm{x})
   D_{\bm{h}_2}l_{[\bm{A},s]}(\bm{v})(\bm{x})
  p(\bm{x}; \bm{\Sigma}(\text{Exp}_{[\bm{A},s]}(\bm{v})))\mathrm{d} \bm{x}
  \\ & = \int_{\mathbb{R}^p} \squared{\bm{x}^T\bracket{\bm{\Sigma}(\text{Exp}_{[\bm{A},s]}(\bm{v}))}^{-1}
  \bracket{D_{\bm{h}_1}\bm{\Sigma}(\text{Exp}_{[\bm{A},s]}(\bm{v}))}
  \bracket{\bm{\Sigma}(\text{Exp}_{[\bm{A},s]}(\bm{v}))}^{-1}\bm{x}}
  \\ & \squared{\bm{x}^T\bracket{\bm{\Sigma}(\text{Exp}_{[\bm{A},s]}(\bm{v}))}^{-1}
  \bracket{D_{\bm{h}_2}\bm{\Sigma}(\text{Exp}_{[\bm{A},s]}(\bm{v}))}
  \bracket{\bm{\Sigma}(\text{Exp}_{[\bm{A},s]}(\bm{v}))}^{-1}\bm{x}}
  p(\bm{x}; \bm{\Sigma}(\text{Exp}_{[\bm{A},s]}(\bm{v})))\mathrm{d} \bm{x}
  \\ & = \tr \curly{\bracket{\bm{\Sigma}(\text{Exp}_{[\bm{A},s]}(\bm{v}))}^{-1}
  D_{\bm{h}_1}\bm{\Sigma}(\text{Exp}_{[\bm{A},s]}(\bm{v}))
  \bracket{\bm{\Sigma}(\text{Exp}_{[\bm{A},s]}(\bm{v}))}^{-1}
  D_{\bm{h}_2}\bm{\Sigma}(\text{Exp}_{[\bm{A},s]}(\bm{v}))}.
  \end{split}
\end{equation} 
Thus, in the particular case $\bm{h} = \bm{h}_1 = \bm{h}_2$:
\begin{equation}\label{eq:r_L2norm}
  \begin{split}
  & \tab[0.4cm] 	\|D_{\bm{h}}r_{[\bm{A},s]}(\bm{v});L^2(\mathbb{R}^p)	\|^{\frac{1}{2}} =
  \int_{\mathbb{R}^p} \squared{D_{\bm{h}}r_{[\bm{A},s]}(\bm{v})(\bm{x})}^2 \mathrm{d} \bm{x}
   \\ & = \frac{1}{4}\int_{\mathbb{R}^p} \bracket{D_{\bm{h}}l_{[\bm{A},s]}(\bm{v})(\bm{x})}^2
  p(\bm{x}; \bm{\Sigma}(\text{Exp}_{[\bm{A},s]}(\bm{v})))\mathrm{d} \bm{x}
  = \frac{1}{2} \tr \curly{\squared{\bracket{\bm{\Sigma}(\text{Exp}_{[\bm{A},s]}(\bm{v}))}^{-1}
  D_{\bm{h}}\bm{\Sigma}(\text{Exp}_{[\bm{A},s]}(\bm{v}))}^2}.
  \end{split}
\end{equation} 
Therefore, $D_{\bm{h}}r_{[\bm{A},s]}(\bm{v}) \in L^2(\mathbb{R}^p)$
and \eqref{eq:r_derivative} is indeed a 
valid Gâteaux derivative at $\bm{v}$ in the direction $\bm{h}$.

To prove that this is indeed the Fréchet derivative,
let us look at the remainder term:

\begin{equation*}
  R(\bm{v}, \bm{h})(\bm{x}) := 
  r_{[\bm{A},s]}(\bm{v}+\bm{h})(\bm{x}) - 
  r_{[\bm{A},s]}(\bm{v})(\bm{x}) - D_{\bm{h}}r_{[\bm{A},s]}(\bm{v})(\bm{x}) 
\end{equation*}
and prove that 
\begin{equation*}
  \lim_{\bm{h} \to \bm{0}} \frac{1}{\abs{\bm{h}}^2}
  \|R(\bm{v}, \bm{h});L^2(\mathbb{R}^p)	\|^2 = 0.
\end{equation*}
To start, the fundamental theorem of calculus shows that

\begin{equation*}
  \begin{split}
  & \tab[0.4cm] r_{[\bm{A},s]}(\bm{v}+\bm{h})(\bm{x}) - 
  r_{[\bm{A},s]}(\bm{v})(\bm{x}) 
  = \int_{0}^1 \frac{d}{dt} r_{[\bm{A},s]}(\bm{v}+t\bm{h})(\bm{x}) \mathrm{d} t 
  \\ & = \int_{0}^1 \lim_{s \to 0} \frac{r_{[\bm{A},s]}(\bm{v}+(t+s)\bm{h})(\bm{x})
  - r_{[\bm{A},s]}(\bm{v}+t\bm{h})(\bm{x})}{s} \mathrm{d} t 
  = \int_{0}^1 D_{\bm{h}}r_{[\bm{A},s]}(\bm{v}+t\bm{h})(\bm{x}) \mathrm{d} t.
  \end{split}
\end{equation*}
Therefore,
by Minkowski's integral inequality and the linearity of $D_{\bm{h}}r_{[\bm{A},s]}(\bm{v})$
in $\bm{h}$, we have
\begin{equation*}
  \begin{split}
  & \tab[0.4cm] 
  \|R(\bm{v}, \bm{h});L^2(\mathbb{R}^p)	\|
  = \|\int_{0}^1 \bracket{D_{\bm{h}}r_{[\bm{A},s]}(\bm{v}+t\bm{h})(\bm{x})- D_{\bm{h}}r_{[\bm{A},s]}(\bm{v})(\bm{x})} 
  \mathrm{d} t;L^2(\mathbb{R}^p)	\|
  \\ & \leqslant \int_{0}^1 
  \|D_{\bm{h}}r_{[\bm{A},s]}(\bm{v}+t\bm{h})- D_{\bm{h}}r_{[\bm{A},s]}(\bm{v})
  ;L^2(\mathbb{R}^p)	\| \mathrm{d} t
  = \abs{\bm{h}} \int_{0}^1 
  \|D_{\frac{\bm{h}}{\abs{\bm{h}}}}r_{[\bm{A},s]}(\bm{v}+t\bm{h})- D_{\frac{\bm{h}}{\abs{\bm{h}}}}r_{[\bm{A},s]}(\bm{v})
  ;L^2(\mathbb{R}^p)	\| \mathrm{d} t
  \end{split}
\end{equation*}
and 
\begin{equation*}
  \begin{split}
  & \tab[0.4cm] \frac{1}{\abs{\bm{h}}^2}
  \|R(\bm{v}, \bm{h});L^2(\mathbb{R}^p)	\|^2 \leqslant
  \curly{\int_{0}^1 
  \|D_{\frac{\bm{h}}{\abs{\bm{h}}}}r_{[\bm{A},s]}(\bm{v}+t\bm{h})- D_{\frac{\bm{h}}{\abs{\bm{h}}}}r_{[\bm{A},s]}(\bm{v})
  ;L^2(\mathbb{R}^p)	\| \mathrm{d} t}^2
  \\ & \leqslant 
  \curly{\int_{0}^1 \sup_{\abs{\bm{u}} = 1}
  \|D_{\bm{u}}r_{[\bm{A},s]}(\bm{v}+t\bm{h})- D_{\bm{u}}r_{[\bm{A},s]}(\bm{v})
  ;L^2(\mathbb{R}^p)	\| \mathrm{d} t}^2.
  \end{split}
\end{equation*}
The proof will be complete once we can show that

\begin{equation}\label{eq:int_sup_u}
  \lim_{\bm{h} \to \bm{0}} 
  \int_{0}^1 \sup_{\abs{\bm{u}} = 1}
  \|D_{\bm{u}}r_{[\bm{A},s]}(\bm{v}+t\bm{h})- D_{\bm{u}}r_{[\bm{A},s]}(\bm{v})
  ;L^2(\mathbb{R}^p)	\| \mathrm{d} t = 0.
\end{equation}
We mus therefore examine the continuity of $D_{\bm{h}}r_{[\bm{A},s]}(\bm{v})$,
which is easy to see if we revisit the formulas.
\eqref{eq:Sigma_derivative} shows that $ D_{\bm{h}}\bm{\Sigma}(\text{Exp}_{[\bm{A},s]}(\bm{v}))$
is smooth in both $\bm{h}$ and $\bm{v}$.
Thus,
\eqref{eq:r_derivative} shows that,
for each fixed $\bm{x} \in \mathbb{R}^p$,
$D_{\bm{h}}r_{[\bm{A},s]}(\bm{v})(\bm{x})$ 
is smooth in both $\bm{h}$ and $\bm{v}$
and \eqref{eq:r_L2norm}
shows that $\|D_{\bm{h}}r_{[\bm{A},s]}(\bm{v});L^2(\mathbb{R}^p)	\|$
is smooth in both $\bm{h}$ and $\bm{v}$.
Combine the two facts, we see that 
$D_{\bm{h}}r_{[\bm{A},s]}(\bm{v})$, as a map to $L^2(\mathbb{R}^p)$,
is continuous in $\bm{h}$ and in $\bm{v}$.
The integrand inside \eqref{eq:int_sup_u}
is thus bounded uniformly 
over all $t \in [0,1]$ and all $\bm{h}$
in a compact neighborhood of $\bm{0}$
and it suffices to show 
that, for every $t \in [0,1]$, that 
\begin{equation}\label{eq:lim_sup_u}
  \lim_{\bm{h} \to \bm{0}} 
  \sup_{\abs{\bm{u}} = 1}
  \|D_{\bm{u}}r_{[\bm{A},s]}(\bm{v}+t\bm{h})- D_{\bm{u}}r_{[\bm{A},s]}(\bm{v})
  ;L^2(\mathbb{R}^p)	\|  = 0,
\end{equation}
thanks to the bounded convergence theorem.
To prove \eqref{eq:lim_sup_u}, 
let $\bm{e}_1, \ldots, \bm{e}_d$ 
be a $g_{[\bm{A},s]}$ orthonormal basis
where $d = pq+1-\frac{q(q-1)}{2}$.
Every unit vector $\bm{u}$ can be 
expressed as $\bm{u} = \sum_{i=1}^d u_i \bm{e}_i$
with $\sum_{i=1}^d u_i^2 = 1$.
By linearity of $D_{\bm{h}}r_{[\bm{A},s]}(\bm{v})$ in $\bm{h}$, 
we have 
\begin{equation*}
  \begin{split}
  & \tab[0.4cm] \|D_{\bm{u}}r_{[\bm{A},s]}(\bm{v}+t\bm{h}) - D_{\bm{u}}r_{[\bm{A},s]}(\bm{v});L^2(\mathbb{R}^p)	\| 
  \leqslant \sum_{i=1}^d \abs{u_i}\|D_{\bm{e}_i}r_{[\bm{A},s]}(\bm{v}+t\bm{h}) - 
  D_{\bm{e}_i}r_{[\bm{A},s]}(\bm{v});L^2(\mathbb{R}^p)	\|
  \\ & \leqslant \sqrt{d} \max_{i=1, \ldots, d} \|D_{\bm{e}_i}r_{[\bm{A},s]}(\bm{v}+t\bm{h}) - 
  D_{\bm{e}_i}r_{[\bm{A},s]}(\bm{v});L^2(\mathbb{R}^p)	\|.
  \end{split}
\end{equation*}
Taking supremum yields
\begin{equation*}
  \sup_{\abs{\bm{u}} = 1}
  \|D_{\bm{u}}r_{[\bm{A},s]}(\bm{v}+t\bm{h})- D_{\bm{u}}r_{[\bm{A},s]}(\bm{v})
  ;L^2(\mathbb{R}^p)	\|  \leqslant 
  \sqrt{d} \max_{i=1, \ldots, d} \|D_{\bm{e}_i}r_{[\bm{A},s]}(\bm{v}+t\bm{h}) - 
  D_{\bm{e}_i}r_{[\bm{A},s]}(\bm{v});L^2(\mathbb{R}^p)	\|.
\end{equation*}
Given that $D_{\bm{e}_i}r_{[\bm{A},s]}(\bm{v}+t\bm{h}) 
\to D_{\bm{e}_i}r_{[\bm{A},s]}(\bm{v})$
in $L^2(\mathbb{R}^p)$
as $\bm{h} \to \bm{0}$
for every $i$, every $t$, and every $\bm{v}$,
the proof of Fréchet differentiability is complete.

Finally,
the formulas \eqref{eq:Sigma_derivative}, \eqref{eq:l_derivative},
and \eqref{eq:r_derivative}
simplifies greatly when $\bm{v} = \bm{0}$:
\begin{equation*}
  \begin{split}
  & \tab[0.4cm] D_{\bm{h}}r_{[\bm{A},s]}(\bm{0})(\bm{x})
  = \frac{1}{2} r_{[\bm{A},s]}(\bm{0})(\bm{x}) D_{\bm{h}}l_{[\bm{A},s]}(\bm{0})(\bm{x})
  \\ & = \frac{1}{2} \sqrt{p(\bm{x}; [\bm{A},s])}
  \tr \curly{\bracket{\bm{\Sigma}([\bm{A},s])}^{-1}
  \squared{\bm{x}\bm{x}^T - \bracket{\bm{\Sigma}([\bm{A},s])}}
  \bracket{\bm{\Sigma}([\bm{A},s])}^{-1} \bracket{D_{\bm{h}}\bm{\Sigma}([\bm{A},s])}}.
  \end{split}
\end{equation*}
The score $s(\bm{x}; [\bm{A},s])(\bm{h})$
therefore equals to $D_{\bm{h}}l_{[\bm{A},s]}(\bm{0})(\bm{x})$
and is indeed given by \eqref{eq:score}.
The Fisher information operator \eqref{eq:G_def} is obtained 
by taking $\bm{v} = \bm{0}$ in \eqref{eq:fisher_computation}.
To show that $G_{[\bm{A},s]}$ is positive definite,
suppose that $\bm{h} \in T_{[\bm{A},s]}M$
is such that $G_{[\bm{A},s]}(\bm{h},\bm{h})=0$.
\eqref{eq:G_def} shows that the Euclidean norm 
of the matrix $\bracket{\bm{\Sigma}([\bm{A},s])}^{-1}
  D_{\bm{h}}\bm{\Sigma}([\bm{A},s])$
  must be zero.
Therefore, $D_{\bm{h}}\bm{\Sigma}([\bm{A},s]) = \bm{0}$
and, by \eqref{eq:Sigma_directional},
it must be the case that $\bm{h} = \bm{0}$.
\end{proof}
